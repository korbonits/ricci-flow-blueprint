\chapter{Overview}

The goal is Hamilton's 1982 theorem: a closed $3$-manifold admitting a metric of
positive Ricci curvature admits a metric of constant positive sectional
curvature. Mathlib carries the terminal statement of the road this sits on, in
\texttt{Mathlib/Geometry/Manifold/PoincareConjecture.lean}.

The obstacle is usually said to be analysis, and eventually it is --- but not
first. Ricci curvature cannot presently be \emph{defined}, for reasons two levels
below the analysis gap, so a project planning around parabolic PDE stalls well
before reaching it. Mathlib has covariant derivatives,
torsion, and metric compatibility on vector bundles, but no parabolic PDE
theory, no elliptic regularity, and no Sobolev spaces of bundle sections ---
so short-time existence of the flow has nothing to stand on. Chapter
\ref{chap:flow} marks that region explicitly. Chapter \ref{chap:homog} is the
branch that can be closed today: on a Lie group the flow is an ODE.

\chapter{Curvature}
\label{chap:curvature}

\begin{definition}[Curvature of a covariant derivative]
  \label{def:curvature}
  \lean{CovariantDerivative.curvature}
  \leanok
  For a covariant derivative $\nab$ on the tangent bundle of $\M$,
  \[ \Riem(X,Y)Z = \nab_X \nab_Y Z - \nab_Y \nab_X Z - \nab_{[X,Y]} Z. \]
\end{definition}

\begin{lemma}[Antisymmetry]
  \label{lem:curvature-antisymm}
  \lean{CovariantDerivative.curvature_antisymm}
  \leanok
  \uses{def:curvature}
  $\Riem(X,Y)Z = -\Riem(Y,X)Z$.
\end{lemma}

\begin{proof}
  \leanok
  Immediate from $[Y,X] = -[X,Y]$ and linearity of $\nab_{(-)} Z$ in the
  direction slot.
\end{proof}

\begin{lemma}[First Bianchi identity]
  \label{lem:bianchi-first}
  \lean{CovariantDerivative.bianchi_first}
  \leanok
  \uses{def:curvature, thm:levi-civita}
  For a torsion-free connection,
  $\Riem(X,Y)Z + \Riem(Y,Z)X + \Riem(Z,X)Y = 0$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved}, for a $C^1$ connection and $C^2$ vector fields, with every
  differentiability side condition discharged.

  Regroup the cyclic sum
  using torsion-freeness on each pair,
  \[ \textstyle\sum \Riem(X,Y)Z = \nab_X [Y,Z] - \nab_{[Y,Z]} X
     + \nab_Y [Z,X] - \nab_{[Z,X]} Y + \nab_Z [X,Y] - \nab_{[X,Y]} Z, \]
  apply torsion-freeness once more to each difference to obtain
  $[X,[Y,Z]] + [Y,[Z,X]] + [Z,[X,Y]]$, and close with Jacobi
  (\texttt{VectorField.leibniz\_identity\_mlieBracket\_apply}).

  Both ingredients are in Mathlib as of August 2026. The $C^2$ hypotheses are
  essential rather than cosmetic: \texttt{torsion\_eq\_zero\_iff} gives
  $\nab_X Y - \nab_Y X = [X,Y]$ only for \texttt{MDiffAt} fields, and the
  second application is to the pair $(X, [Y,Z])$, so the bracket itself must be
  differentiable.

  (An earlier version of this blueprint claimed the identity was blocked on
  $Z$-slot tensoriality and \texttt{mkHom₃}. Those are prerequisites for
  packaging curvature as a \emph{tensor}, not for the operator identity stated
  here.)
\end{proof}

\begin{lemma}[The Lie bracket is a derivation]
  \label{lem:derivation}
  \lean{VectorField.mlieBracket_apply_fun}
  \leanok
  $[V,W]f = V(Wf) - W(Vf)$, for $f$ of class $C^2$ at the point and $V,W$
  differentiable there, on any manifold --- boundary and corners allowed.
\end{lemma}

\begin{proof}
  \leanok
  Absent from Mathlib in both the vector-space and the manifold setting. Over a
  vector space the two second-derivative terms cancel by symmetry of the second
  derivative, leaving exactly the bracket. Transporting to a manifold turns on one
  observation: $\mathrm{d}f$ at a nearby point $y$ is defined through the chart
  \emph{at $y$}, so one first re-expresses it through the single chart at $x$,
  uniformly over that chart's source. The outer derivative is then an ordinary
  \texttt{fderivWithin} and the vector-space identity transports verbatim, with
  $\mathrm{range}\,I$ supplying the side conditions the \texttt{Within} calculus
  needs. No manifold-level second-derivative symmetry is required.
\end{proof}

\begin{lemma}[Tensoriality of the curvature operator]
  \label{lem:tensoriality}
  \lean{CovariantDerivative.curvature_smul_left}
  \leanok
  \uses{def:curvature, lem:derivation}
  $\Riem$ is $C^\infty(\M)$-linear in each slot.
\end{lemma}

\begin{proof}
  \leanok
  First slot: expanding $\nab_{fX}Z = f\,\nab_X Z$ leaves a $(Yf)\,\nab_X Z$ term
  from Leibniz, and $[fX,Y] = f[X,Y] - (Yf)X$ leaves a matching one from the
  bracket, with opposite sign; they cancel. The second slot follows by
  antisymmetry. The third leaves a residue $[\,X(Yf) - Y(Xf) - [X,Y]f\,]\,Z$,
  which vanishes by Lemma~\ref{lem:derivation}.
\end{proof}

\chapter{The Levi-Civita connection}
\label{chap:levi-civita}

\begin{theorem}[Existence and uniqueness]
  \label{thm:levi-civita}
  \lean{RicciFlowBlueprint.exists_unique_leviCivita}
  There is exactly one torsion-free covariant derivative compatible with a
  given Riemannian metric $\g$.
\end{theorem}

\begin{proof}
  \textbf{In flight upstream:} mathlib4 PR \#36845 supplies the Koszul formula,
  uniqueness, the construction, and the compatibility/torsion-free proof. It has
  been open since March 2026 and is \texttt{awaiting-author}. Do not duplicate
  it.

  Uniqueness from the Koszul formula; existence by constructing $\nab$ from it.
  Mathlib supplies both ingredients as of August 2026:
  \texttt{CovariantDerivative.IsMetricCompatible} in
  \texttt{CovariantDerivative/Metric.lean} and
  \texttt{CovariantDerivative.torsion} in \texttt{.../Torsion.lean}. Note the
  metric must reach the tangent bundle through \texttt{RiemannianBundle}
  rather than a raw \texttt{[$\forall$ x, InnerProductSpace $\R$ (TangentSpace I x)]}
  binder, which would create a topology diamond.
\end{proof}

\begin{definition}[Ricci and scalar curvature]
  \label{def:ricci}
  \uses{def:curvature, thm:levi-civita}
  $\Ric(X,Y) = \operatorname{tr}(Z \mapsto \Riem(Z,X)Y)$ and
  $\scal = \operatorname{tr}_\g \Ric$.

  \textbf{Mathematics done; packaging outstanding.} The trace needs $\Riem$ to
  descend from vector \emph{fields} to vectors, and the criterion for that is
  $C^\infty(\M)$-linearity in each slot --- Lemma~\ref{lem:tensoriality}. What
  remains is construction rather than mathematics: assembling a pointwise tensor
  from that linearity and tracing it, for which \texttt{TensorialAt} supplies
  only first-order hypotheses and \texttt{mkHom₃} does not exist. An earlier
  version of this blueprint recorded the obstruction as mathematical. That was
  wrong, and the error mattered: it put the wall two levels above where it is.
\end{definition}

\chapter{Ricci flow on left-invariant metrics}
\label{chap:homog}

This is the branch with no PDE in it. A left-invariant metric on a Lie group
$G$ is determined by an inner product on $\mathfrak{g}$, so
$\partial_t \g = -2\Ric(\g)$ becomes an ODE on a finite-dimensional space, and
Mathlib's Picard--Lindel\"of suffices.

\begin{lemma}
  \label{lem:ricci-lie-ode}
  \uses{def:ricci}
  On a Lie group, the Ricci endomorphism of a left-invariant metric is a
  smooth function of the metric alone; the normalized flow is a locally
  Lipschitz vector field on the space of inner products on $\mathfrak{g}$.
\end{lemma}

\begin{theorem}
  \label{thm:ricci-lie-shortTime}
  \uses{lem:ricci-lie-ode}
  The left-invariant Ricci flow has a unique maximal solution through each
  initial metric.
\end{theorem}

\chapter{Ricci flow on a closed manifold}
\label{chap:flow}

Everything below is blocked on analysis Mathlib does not have. It is stated so
the dependency graph is honest, not because it is close.

\begin{definition}[Ricci flow]
  \label{def:ricci-flow}
  \uses{def:ricci}
  A smooth family $\g(t)$ of metrics with $\partial_t \g = -2\Ric(\g)$.
\end{definition}

\begin{theorem}[Short-time existence]
  \label{thm:short-time}
  \uses{def:ricci-flow}
  On a closed manifold, Ricci flow has a unique solution on a maximal interval
  $[0,T)$ for any smooth initial metric.

  \textbf{Blocked.} Requires quasilinear parabolic systems on sections of
  vector bundles (via DeTurck's trick, or Nash--Moser in Hamilton's original),
  and hence Sobolev spaces of bundle sections and elliptic regularity. None of
  this exists in Mathlib.
\end{theorem}

\begin{theorem}[Hamilton 1982]
  \label{thm:hamilton}
  \uses{thm:short-time, lem:bianchi-first}
  A closed $3$-manifold admitting a metric of positive Ricci curvature admits a
  metric of constant positive sectional curvature.

  \textbf{Blocked.} Additionally requires the tensor maximum principle,
  the pinching estimates, Shi's derivative estimates, and convergence of the
  normalized flow.
\end{theorem}
